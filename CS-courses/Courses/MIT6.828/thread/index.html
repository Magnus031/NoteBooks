// per-process data for the trap handling code in trampoline.S.
// sits in a page by itself just under the trampoline page in the
// user page table. not specially mapped in the kernel page table.
// uservec in trampoline.S saves user registers in the trapframe,
// then initializes registers from the trapframe's
// kernel_sp, kernel_hartid, kernel_satp, and jumps to kernel_trap.
// usertrapret() and userret in trampoline.S set up
// the trapframe's kernel_*, restore user registers from the

  /*  88 */ uint64 t2;

  /* 104 */ uint64 s1;

  /* 136 */ uint64 a3;
  /* 144 */ uint64 a4;

    首先是 上下文， 什么是上下文其实就是 ra 和 sp 两个寄存器以及其他的 callee registers s0-s11 一共 14 个寄存器。

    • 为什么要存在 ra? 而不是 pc

      • 最本质的原因 就是因为 swtch 是一个 函数调用，而不是一个 jmp 指令，所以我们需要保存 ra 的值。
      • pc 表示的是当前的指令地址，而 ra 表示的是当前函数调用的返回地址。
      • 上下文切换 的目的就是 恢复另一个内核线程的上下文, 当 swtch 返回的时候，返回的地方并不是调用它的 scheduler 处，而是新的线程的调用链中
      • 而 pc 是一个寄存器，保存的是 指令地址，并非是函数调用的返回地址。

    • 为什么要保存 sp?

      • 这个也好理解，sp 是函数调用的栈顶指针，保存了当前函数调用的栈顶地址。
      • 当我们进行上下文切换的时候，我们需要恢复这个栈顶地址，来保证函数调用的正确性。
      • 每个内核线程中都有自己独立的内核栈，当前的内核线程的 函数调用链和局部变量不能被破坏

    • 为什么要保存 s0-s11?

      • 这些寄存器是 callee-saved registers, 当我们进行函数调用的时候，调用者需要保证这些寄存器的值不变。
      • 所以在进行上下文切换的时候，我们需要保存这些寄存器的值，以便在恢复上下文的时候能够正确地恢复这些寄存器的值。

    // push_off/pop_off are like intr_off()/intr_on() except that they are matched:
// it takes two pop_off()s to undo two push_off()s.  Also, if interrupts
// are initially off, then push_off, pop_off leaves them off.

void
push_off(void)
{
  int old = intr_get();

  intr_off();
  if(mycpu()->noff == 0)
    mycpu()->intena = old;
  mycpu()->noff += 1;
}

void
pop_off(void)
{
  struct cpu *c = mycpu();
  if(intr_get())
    panic("pop_off - interruptible");
  if(c->noff < 1)
    panic("pop_off");
  c->noff -= 1;
  if(c->noff == 0 && c->intena)
    intr_on();
}

// enable device interrupts
static inline void
intr_on()
{
  w_sstatus(r_sstatus() | SSTATUS_SIE);
}

// disable device interrupts
static inline void
intr_off()
{
  w_sstatus(r_sstatus() & ~SSTATUS_SIE);
}

// are device interrupts enabled?
static inline int
intr_get()
{
  uint64 x = r_sstatus();
  return (x & SSTATUS_SIE) != 0;
}

    其实我们看上述的 push_off 和 pop_off 的实现，其实是基于 sstatus 的中断机制来进行的。我们看 documentation 中是这样说的:

    The SIE bit enables or disables all interrupts in supervisor mode. When SIE is clear, interrupts are not taken while in supervisor mode. When the hart is running in user-mode, the value in SIE is ignored, and supervisor-level interrupts are enabled. The supervisor can disable individual interrupt sources using the sie CSR. The SPIE bit indicates whether supervisor interrupts were enabled prior to trapping into supervisor mode. When a trap is taken into supervisor mode, SPIE is set to SIE, and SIE is set to 0. When an SRET instruction is executed, SIE is set to SPIE, then SPIE is set to 1.

    解决的问题

    在多核 CPU 环境下，通常每个CPU在使用 critical section的时候，通常是持有一把锁的。但是当 timer interrupt 在获取竞态资源的时候，CPU 可能会被打断，这个时候就会导致 CPU 在持有锁的情况下被打断，这个时候就会导致死锁的情况发生。所以 acquire 操作会在进入 critical section 之前关闭 interrupt 状态，就可以避免这种情况的发生。

    但是存在了下面这个问题:

    This is a bug that could lead to a deadlock. The intr_on() call in the nested function function_B incorrectly re-enabled interrupts, violating the assumption of the outer function function_A.

    就是说在函数调用的时候，被调用的那个函数如果不小心的开启了中断，那么就会导致死锁的情况发生。原先的函数调用的假设就被打破了。所以这里我们用一个 push_off() 和 pop_off() 来进行处理这个问题。

    我们来看上述的逻辑，其实就是利用了两个 cpu->noff 和 cpu->intena 来进行处理. 正如上述所说， cpu->noff 保证的是被嵌套的次数，而 cpu->intena 保证的是之前的中断状态。用来恢复， push_off 和 pop_off 是相对应的，所以这里以 push_off 为例:

    • 先获取当前的中断状态 old = intr_get()
    • 检查当前的 cpu->noff 是否为 0，如果是 0，那么就将 cpu->intena 设置为之前的中断状态 old，用来后续恢复状态
    • 我们将 cpu->noff 加 1，记录的是 how many push_off() has been called.

    // Acquire the lock.
// Loops (spins) until the lock is acquired.
void
acquire(struct spinlock *lk)
{
  push_off(); // disable interrupts to avoid deadlock.
  if(holding(lk))
    panic("acquire");

  // On RISC-V, sync_lock_test_and_set turns into an atomic swap:
  //   a5 = 1
  //   s1 = &lk->locked
  //   amoswap.w.aq a5, a5, (s1)
  while(__sync_lock_test_and_set(&lk->locked, 1) != 0)
    ;

  // Tell the C compiler and the processor to not move loads or stores
  // past this point, to ensure that the critical section's memory
  // references happen strictly after the lock is acquired.
  // On RISC-V, this emits a fence instruction.
  __sync_synchronize();

  // Record info about lock acquisition for holding() and debugging.
  lk->cpu = mycpu();
}

// Release the lock.
void
release(struct spinlock *lk)
{
  if(!holding(lk))
    panic("release");

  lk->cpu = 0;

  // Tell the C compiler and the CPU to not move loads or stores
  // past this point, to ensure that all the stores in the critical
  // section are visible to other CPUs before the lock is released,
  // and that loads in the critical section occur strictly before
  // the lock is released.
  // On RISC-V, this emits a fence instruction.
  __sync_synchronize();

  // Release the lock, equivalent to lk->locked = 0.
  // This code doesn't use a C assignment, since the C standard
  // implies that an assignment might be implemented with
  // multiple store instructions.
  // On RISC-V, sync_lock_release turns into an atomic swap:
  //   s1 = &lk->locked
  //   amoswap.w zero, zero, (s1)
  __sync_lock_release(&lk->locked);

  pop_off();
}

    __sync_lock_test_and_set 是一个内置的 gcc 函数，目的是用来原子交换的。我们需要保证在设置锁的时候不被打断, amoswap.w.aq a5, a5, (s1) 这个指令就是用来实现原子交换的。

    __sync_synchronize() 核心目的是提供了一个 full memory barrier “在屏障之前的所有内存操作(loads and stores)，都必须在屏障之后的所有内存操作之前完成。”

    • 避免编译器指令重排，因为为了优化性能，编译器可能会改变代码中的指令顺序
    • 避免 CPU 指令重拍，也就是 乱序, 因为在流水线中会出现多级指令并行执行的情况，可能会导致指令的执行顺序与代码中的顺序不一致。我们通常会通过乱序来进行优化性能。

    而 __sync_synchronize() 的作用就是保证了在 acquire 和 release 之间的所有内存操作都不会被重排。

    • Visibility 确保在屏障之前的写操作，对所有其他CPU核心都是可见的

      • 为什么是 可见性？ 因为要保证在一个CPU的所有写的操作，别的CPU是能看到这个写操作的结果的。使得在竞态资源写操作结束的时候 拿到的是最新的数据

    • Ordering 防止编译器和CPU对内存操作进行重排，确保在屏障之前的所有操作都在屏障之后的操作之前完成。

    void
scheduler(void)
{
  struct proc *p;
  struct cpu *c = mycpu();

  c->proc = 0;
  for(;;){
    // The most recent process to run may have had interrupts
    // turned off; enable them to avoid a deadlock if all
    // processes are waiting.
    intr_on();

    for(p = proc; p < &proc[NPROC]; p++) {
      acquire(&p->lock);
      if(p->state == RUNNABLE) {
        // Switch to chosen process.  It is the process's job
        // to release its lock and then reacquire it
        // before jumping back to us.
        p->state = RUNNING;
        c->proc = p;
        swtch(&c->context, &p->context);

        // Process is done running for now.
        // It should have changed its p->state before coming back.
        c->proc = 0;
      }
      release(&p->lock);
    }
  }
}

    1. intr_on() 的目的

    intr_on() 在 scheduler 中的作用是避免死锁，而不是为了每一个进程去开启中断。

    • 死锁场景：假设所有进程都因为某种原因（例如，等待 I/O 完成）进入了 SLEEPING 状态。并且，最后一个运行的进程在陷入内核时，中断是关闭的。
    • 后果：scheduler 循环遍历进程列表，找不到任何 RUNNABLE 的进程。它会一直循环，但由于中断是关闭的，时钟中断不会发生，I/O 完成中断也不会发生，所有进程都将永远无法被唤醒，系统会陷入死锁。

    为了防止这种情况，scheduler 必须确保在找不到任何可运行进程时，中断是开启的，以便时钟或其他设备中断能够唤醒某个进程。

    2. 为什么在外部循环？

    scheduler 的 for(;;) 循环是调度器线程的执行主体。intr_on() 放在外部循环中，保证了调度器线程本身在大部分时间里是开启中断的。

    • 如果放在内部循环：

void
sleep(void *chan, struct spinlock *lk)
{
  struct proc *p = myproc();

  // Must acquire p->lock in order to
  // change p->state and then call sched.
  // Once we hold p->lock, we can be
  // guaranteed that we won't miss any wakeup
  // (wakeup locks p->lock),
  // so it's okay to release lk.

  acquire(&p->lock);  //DOC: sleeplock1
  release(lk);

  // Go to sleep.
  p->chan = chan;
  p->state = SLEEPING;

  sched();

  // Tidy up.
  p->chan = 0;

  // Reacquire original lock.
  release(&p->lock);
  acquire(lk);
}

// Wake up all processes sleeping on chan.
// Must be called without any p->lock.
void
wakeup(void *chan)
{
  struct proc *p;

  for(p = proc; p < &proc[NPROC]; p++) {
    if(p != myproc()){
      acquire(&p->lock);
      if(p->state == SLEEPING && p->chan == chan) {
        p->state = RUNNABLE;
      }
      release(&p->lock);
    }
  }
}

    chan 的含义

    chan(channel) 是一个任意的内存地址，它被用作一个 同步点， 它本身不指向任何的数据结构，但是它的地址可以唯一地标识一个特定的等待事件。

    • sleep : 一个线程想要等待某个特定的事件发生的时候，就会调用 sleep 并且传入一个 chan。它就会进行 SLEEPING 状态，并且记录下自己正在等待的这个chan
    • wakeup : 当某个事件发生的时候，另一个线程会调用 wakeup 并传入相同的 chan 地址，然后 wakeup 会遍历所有正在睡眠的线程，找到那些正在等待这个特定 chan 的线程把它们叫醒.

V(struct semaphore *s)
{
  acquire(&s->lock);
  s->count ++;
  // important; 及时唤醒那些等待的线程
  // 为什么传入的是 `s` 而不是 `&s->lock`?
  // 因为 `wakeup` 需要知道是哪个信号量被释放了，而不是哪个锁被释放了。
  wakeup(s);
  release(&s->lock);
}

void
P(struct semaphore *s)
{
  acquire(&s->lock);
  while(s->count == 0) {
    // 如果信号量的计数为 0，那么就需要等待
    sleep(s, &s->lock);
  }
  s->count --;
  release(&s->lock);
}

    上述的代码中传入的 sleep(s, &s->lock) 中的 s 就是 chan，它是一个任意的内存地址，用来标识这个等待事件。wakeup(s) 就是唤醒所有等待这个 chan 的线程。而传入的这个 &s->lock, 就是为了先释放掉，因为我们进行 P 操作的时候刚开始就获取了锁 acquire(&s->lock)，所以在 sleep 的时候需要先释放掉这个锁，避免死锁的情况发生。

    进程间的通信

    首先，我们知道的是 进程和线程的区别就是，前者是资源配置的基本单元，而后者是CPU任务调度的基本单元

    所以也就是说，二者是有区别的，前者是相互隔离，并不知道对方的存在，同时对于每一个进程来说，都配备一个单独的地址空间。而后者是共享同一个地址空间的，比如一个进程中可以有多个线程，而线程之间是共享 内存，他们只有独立的寄存器和栈。

    所以对于二者的通信方式，我们其实很明显的知道差别了

    进程之间的通信方式就有几种:

    1. 管道(pipe) : 通过管道来进行通信，管道是一个特殊的文件，它可以在两个进程之间传递数据。
    2. 信号(signal) : 信号是一种异步的通知机制，可以用来通知进程发生了某个事件，比如中断、异常等。
    3. 消息队列(message queue) : 通过消息队列来进行通信，消息队列是一个特殊的内存区域，可以在多个进程之间传递数据。
    4. 共享内存(shared memory) : 通过共享内存来进行通信，多个进程可以访问同一块内存区域，从而实现数据的共享。

    重点来看 3,4 其实就是一种 trade-off 的方式，前者是通过内核来进行通信，而后者是通过共享内存来进行通信。论速度，共享内存的速度是最快的，因为它不需要经过内核的上下文切换。但是共享内存需要进行同步，避免数据竞争。而消息队列则是通过内核来进行通信，速度相对较慢，但是不需要进行同步, 慢在多次拷贝 (用户态->内核态->用户态)。
