    #!/usr/bin/env python3

import fcntl
import struct
import os
import time
from scapy.all import *

TUNSETIFF = 0x400454ca
IFF_TUN   = 0x0001
IFF_TAP   = 0x0002
IFF_NO_PI = 0x1000

# Create the tun interface
tun = os.open("/dev/net/tun", os.O_RDWR)
ifr = struct.pack('16sH', b'wei%d', IFF_TUN | IFF_NO_PI)
ifname_bytes  = fcntl.ioctl(tun, TUNSETIFF, ifr)

ifname = ifname_bytes.decode('UTF-8')[:16].strip("\x00")

# start the tun interface
os.system("ip addr add 192.168.54.99/24 dev {}".format(ifname))
os.system("ip link set dev {} up".format(ifname))

# build for the udp socket
IP_A = "0.0.0.0"
PORT = 9090

sock = socket.socket(socket.AF_INET,socket.SOCK_DGRAM)
sock.bind((IP_A,PORT))

while True:
    data, (ip,port) = sock.recvfrom(2048)
    # 打印ip, port, IP_A PORT
    print("{}:{} --> {}:{}".format(ip,port,IP_A,PORT))
    pkt = IP(data)
    # 打印 package 的源和终点
    print("SubNetwork: {} --> {}".format(pkt.src,pkt.dst))
    # 通过我们创建的 tun 接口进行
    os.write(tun,bytes(pkt))
    print("已经完成了将数据包通过 TUN 接口 {} 进行转发".format(ifname))

    #!/usr/bin/env python3

import fcntl
import struct
import os
import time
from scapy.all import *

TUNSETIFF = 0x400454ca
IFF_TUN   = 0x0001
IFF_TAP   = 0x0002
IFF_NO_PI = 0x1000

# Create the tun interface
tun = os.open("/dev/net/tun", os.O_RDWR)
ifr = struct.pack('16sH', b'wei%d', IFF_TUN | IFF_NO_PI)
ifname_bytes  = fcntl.ioctl(tun, TUNSETIFF, ifr)

ifname = ifname_bytes.decode('UTF-8')[:16].strip("\x00")

# start the tun interface
os.system("ip addr add 192.168.54.99/24 dev {}".format(ifname))
os.system("ip link set dev {} up".format(ifname))
os.system("ip route add 192.168.53.0/24 dev {} via 192.168.54.99".format(ifname))

# build for the udp socket
IP_A = "0.0.0.0"
PORT = 9090

sock = socket.socket(socket.AF_INET,socket.SOCK_DGRAM)
sock.bind((IP_A,PORT))


client_addr = None
while True:
    # 捕获这个数据包;
   ready, _ , _ = select.select([sock,tun],[],[])
   for fd in ready:
      if fd is sock:
            data ,(ip,port) = sock.recvfrom(2048)
            pkt = IP(data)
            print("From Client : {} --> {}".format(pkt.src,pkt.dst))
            client_addr = (ip,port)
            os.write(tun,data) 


      if fd is tun:
            packet = os.read(tun,2048)
            pkt = IP(packet)
            print("From tun ==>: {} --> {}".format(pkt.src,pkt.dst))
            # 进行回复
            if client_addr:             
                sock.sendto(packet,client_addr)
                print("发送到服务器:{} -->{}".format(pkt.src,pkt.dst))

    数据包如何流动的?

    • Step1 : 我们在主机U 进行了 ping 192.168.60.5 操作，测试连通性，此时会发送一个 echo request 的请求。
    • Step2 : 这个数据包会被 tun_client.py 中的 tun(192.168.53.99) 接口捕获到，并且进行一个解析。
    • Step3 : 因为我们进行了增加路由的操作，那么这个数据包会被路由到 tun_client.py 中的 sock 中，并且发送到 server(192.168.54.99) 中。
    • Step4 : server 中的 sock 接口会捕获到这个数据包，并且进行一个解析。
    • Step5 : server 中的 tun 接口会捕获到这个数据包，并且进行一个解析。
    • Step6 : 主机V接受到了这个数据包，并且进行一个 echo reply 的回应。
    • Step7 : server 中的 tun 接口会捕获到这个数据包，并且进行一个解析。
    • Step8 : server 中的 sock 接口会捕获到这个数据包，并且进行一个解析。
    • Step9 : client 中的 sock 接口会捕获到这个数据包，并且进行一个解析。
    • Step10 : client 中的 tun 接口会捕获到这个数据包，并且进行一个解析。
    • Step11 : 主机 V 接受到了，此刻说明联通了。

    #!/usr/bin/env python3

import fcntl
import struct
import os
import time
from scapy.all import *

TUNSETIFF = 0x400454ca
IFF_TUN   = 0x0001
IFF_TAP   = 0x0002
IFF_NO_PI = 0x1000

# Create the tun interface
tap = os.open("/dev/net/tun", os.O_RDWR)
ifr = struct.pack('16sH', b'zhou%d', IFF_TAP | IFF_NO_PI)
ifname_bytes  = fcntl.ioctl(tap, TUNSETIFF, ifr)

# Get the interface name
ifname = ifname_bytes.decode('UTF-8')[:16].strip("\x00")
print("Interface Name: {}".format(ifname))

# Open the Interface;
os.system("ip link set dev {} up".format(ifname))
os.system("ip addr add 192.168.53.99/24 dev {}".format(ifname))


while True:
   # 捕获这个数据包;
   packet = os.read(tap,2048)
   if packet:
       ether = Ether(packet)
       print(ether.summary())
       FAKE_MAC="aa:bb:cc:dd:ee:ff"
       if ARP in ether and ether[ARP].op == 1:
           arp = ether[ARP]
           newether = Ether(dst=ether.src,src = FAKE_MAC)
           newarp = ARP(psrc=arp.pdst,hwsrc=FAKE_MAC,pdst=arp.psrc,hwdst=ether.src,op=2)
           newpkt = newether/newarp
           print("****** Fakeresponse : {}".format(newpkt.summary()))
           os.write(tap,bytes(newpkt))
